.github/workflows/ci.yml (Multi-Stage DAG Specification) 
name: HuntYourTrack CI/CD DevOps Pipeline

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  # Stage 1: Parallel Standards Check
  code-standards:
    name: 📐 1. Architecture & Asset Standards
    runs-on: ubuntu-latest

  # Stage 2: Parallel Security Check
  security-audit:
    name: 🛡️ 2. Security & Zero-Contamination Audit
    runs-on: ubuntu-latest

  # Stage 3: Playwright Smoke (DAG Junction)
  browser-smoke-tests:
    name: 🧪 3. Playwright Browser Smoke Suite
    needs: [code-standards, security-audit]
    runs-on: ubuntu-latest

  # Stage 4: Team Velocity & Deadlines
  team-metrics:
    name: 📊 4. Team Velocity & Deadline Monitor
    needs: [browser-smoke-tests]
    runs-on: ubuntu-latest

  # Stage 5: Executive Signoff
  quality-gate-signoff:
    name: 🚀 5. Executive Quality Gate & Visual Dashboard
    needs: [team-metrics]
    runs-on: ubuntu-latest
